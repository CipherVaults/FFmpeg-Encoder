const root=__dirname;
test('literal B-only scope preserves corrected driver except reviewed scope values',()=>{let s=fs.readFileSync(path.join(root,'../ffmpeg-aac-range090-bound-attempt1/driver.cjs'),'utf8');s=s.replace(" Object.freeze({id:'A',seconds:8,width:1280,height:720,fps:30,resolution:'source',startUs:613000,endUs:2721000,frequency:523}),\n",'').replace('inspect:2','inspect:4').replace('totalMs:1200000','totalMs:600000').replace('maxChildren:14','maxChildren:16').replace('approved-two-aac-range-cases','approved-one-aac-range-B-case').replace('passed-two-experimental-cells-not-activation','passed-one-B-experimental-cell-not-activation');assert.equal(fs.readFileSync(path.join(root,'driver.cjs'),'utf8'),s);assert(fs.readFileSync(path.join(root,'frozen-checks.cjs')).equals(fs.readFileSync(path.join(root,'../ffmpeg-aac-range090-bound-attempt1/frozen-checks.cjs'))));assert.equal(driver.SPECS.length,1);assert.deepEqual(driver.SPECS[0],{id:'B',seconds:31,width:1920,height:1080,fps:30,resolution:'720',startUs:28013000,endUs:30087000,frequency:617});assert.equal(Object.values(driver.LEDGER).reduce((a,b)=>a+b,0),16);});
async function scheduled(id,limit){const saved=JSON.parse(fs.readFileSync(path.join(root,`../ffmpeg-aac-range090-owned-attempt1/${id}/report.json`)));let requests=1,spawned=1;const intervals=[];const result=await finishInspection('/inert/source.mp4',saved.inspection.probe,{ffprobe:'/inert/ffprobe',run:async(executable,args)=>{requests++;assert(requests<=limit);spawned++;intervals.push(args[args.indexOf('-read_intervals')+1]);return {stdout:JSON.stringify({frames:[]}),stderr:''};}});return {requests,spawned,intervals,result};}
test('actual frozen inspector schedules start only for8seconds and start/middle/end for31',async()=>{const a=await scheduled('A',4),b=await scheduled('B',4);assert.deepEqual(a.intervals,['0.000000%2.000000']);assert.deepEqual(b.intervals,['0.000000%2.000000','15.500000%16.500000','29.000000%31.000000']);assert.equal(a.spawned,2);assert.equal(b.spawned,4);assert.equal(b.requests,4);});
test('old ledger reproduces refused-before-spawn attempts, not four actual children',async()=>{const b=await scheduled('B',2);assert.equal(b.requests,4);assert.equal(b.spawned,2);assert.equal(b.result.summary.sampling[1].code,'ERR_ASSERTION');assert.equal(b.result.summary.sampling[2].code,'ERR_ASSERTION');assert.equal(b.spawned+1,3);});
test('fresh bound owner/gate do not exist, old evidence remains separate',()=>{const b=require('./BINDING.json');assert(!fs.existsSync(b.ownerRoot));assert(!fs.existsSync(b.gatePath));assert(b.ownerRoot.endsWith('ffmpeg-aac-range090-owned-B-attempt2'));assert.equal(require('../ffmpeg-aac-range090-owned-attempt1/summary.json').status,'failed-retained');});
