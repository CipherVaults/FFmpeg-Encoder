const {refreshBoundary}=require('./continuation.cjs');
test('refresh strictly follows awaited owner and runtime; failures prevent continuation',async()=>{for(const stage of ['owner','runtime','refresh',null]){const events=[];const go=n=>async()=>{await Promise.resolve();events.push(n);if(stage===n)throw Error(n);};let admitted=false;try{await refreshBoundary({check:()=>{},ownerCheck:go('owner'),runtimeCheck:go('runtime'),refresh:go('refresh')});admitted=true;}catch{}assert.equal(admitted,stage===null);assert.deepEqual(events,stage==='owner'?['owner']:stage==='runtime'?['owner','runtime']:['owner','runtime','refresh']);}});
