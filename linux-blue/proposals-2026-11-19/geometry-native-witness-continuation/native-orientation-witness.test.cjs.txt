'use strict';
const test=require('node:test'),assert=require('node:assert/strict'),fs=require('node:fs'),path=require('node:path'),Module=require('node:module');
const base=path.resolve(__dirname,'..');
function load(file,replace={}){const m=new Module(file,module);m.filename=file;m.paths=Module._nodeModulePaths(path.dirname(file));const original=m.require.bind(m);m.require=n=>Object.hasOwn(replace,n)?replace[n]:original(n);m._compile(fs.readFileSync(file,'utf8'),file);return m.exports;}
function setup(){let now=0,closed=false,mutate=()=>{},reads=0;let identity;
 const witness=load(path.join(base,'src/native-orientation-witness.cjs'),{
 'node:perf_hooks':{performance:{now:()=>now}},
 './source-identity.cjs':{readSourceIdentity:async()=>{reads++;return structuredClone(identity);}},
 './remux-edit-lists.cjs':{observeRemuxSourcePlayback:async o=>{assert.equal(o.maxRecords,20000);assert.equal(o.target,'mp4');assert(o.timeoutMs<=120000);o.onProgress();const r={complete:true,identity:structuredClone(identity),tracks:o.probe.streams.map(s=>({outputIndex:s.index,trackId:Number(BigInt(s.id)),controls:{matrix:[65536,0,0,0,65536,0,0,0,1073741824]},dimensions:{width:(s.width||0)*65536,height:(s.height||0)*65536}}))};await mutate(r);closed=true;return r;}}});
 const family=load(path.join(base,'src/operation-family.cjs'),{'./native-orientation-witness.cjs':witness});
 const f=path.join(__dirname,'operation-family-square-geometry.test.cjs');let text=fs.readFileSync(f,'utf8').split('for(const [w,h,ow,oh]')[0];text+='\nmodule.exports={probe,family,CPU,BUDGET,SOURCE,RUNTIME};';const m=new Module(f,module);m.filename=f;m.paths=Module._nodeModulePaths(__dirname);m._compile(text,f);const x=m.exports;
 identity=structuredClone(x.SOURCE);const probe=x.probe({absent:true});Object.assign(probe.streams[0],{width:960,height:1440,display_aspect_ratio:'2:3',id:'0x1'});probe.streams[1].id='0x2';const inspection={probe,color:require('../src/media.cjs').classifyColor(probe),duration:4};
 const selector=new family.OperationFamilySelector({policyIdentity:'private-native-test',families:[{...x.family(),id:'short-sdr-range-square-geometry-v1'}]});const request=require('../src/operation-contract.cjs').resolveFileRequest({fileId:'private',identityRevision:1,draftRevision:1,operation:'encode',overrides:{range:{startUs:1000000,endUs:2000000},resolution:'720',audioPolicy:'precision-aac'}});
 const observe=()=>witness.observeNativeOrientation({file:identity.canonicalPath,inspection,sourceIdentity:identity,runtimeIdentity:x.RUNTIME,timeoutMs:1000});
 const compose=token=>{const orientation=selector.orientationQualification({inspection,request,runtimeIdentity:x.RUNTIME,settings:x.CPU,sourceIdentity:identity,nativeOrientationWitness:token});const plan=require('../src/encode-operation.cjs').buildEncodeOperation({input:identity.canonicalPath,output:path.resolve('synthetic-output.mkv'),inspection,request,runtimeIdentity:x.RUNTIME,settings:x.CPU,...orientation});const bindings={plan,settings:x.CPU,sourceIdentity:identity,sourceProbeFingerprint:require('../src/encode-operation.cjs').fingerprintEncodeSource(probe),runtimeIdentity:x.RUNTIME,environment:null,budget:x.BUDGET,executionBudgetMs:60000,nativeOrientationWitness:token};return {plan,bindings,token:selector.select(bindings)};};
 return {witness,observe,compose,selector,inspection,identity,setMutation:f=>mutate=f,setNow:t=>now=t};}
test('actual selector/build/currentness consumes private issued native witness; raw probe unchanged',async()=>{const x=setup(),before=JSON.stringify(x.inspection);const w=await x.observe(),c=x.compose(w);assert.equal(c.plan.contract.geometry.output.width,720);assert.equal(c.plan.contract.geometry.output.height,1080);assert(x.selector.requireCurrent(c.token,c.bindings));assert.equal(JSON.stringify(x.inspection),before);x.witness.revokeNativeOrientation(w);assert.throws(()=>x.selector.requireCurrent(c.token,c.bindings));});
test('unwitnessed and copied arbitrary records never admit absent orientation',async()=>{const x=setup();assert.throws(()=>x.compose(undefined));const w=await x.observe();assert.throws(()=>x.compose({...w}));});
test('native rotated matrix and foreign track refuse without issuing token',async()=>{for(const mutate of [r=>r.tracks[0].controls.matrix[0]=0,r=>r.tracks[0].trackId=9,r=>r.identity.ino='99',r=>{throw Error('unrelated edit refusal');}]){const x=setup();x.setMutation(mutate);await assert.rejects(x.observe());}});
test('delayed observer expiry and source replacement refuse',async()=>{for(const mutate of [x=>x.setNow(1001),x=>x.identity.ino='99']){const x=setup();x.setMutation(()=>mutate(x));await assert.rejects(x.observe());}});
test('reported rotation cannot be overwritten by genuine witness',async()=>{const x=setup(),w=await x.observe();x.inspection.probe.streams[0].tags={rotate:'90'};assert.throws(()=>x.compose(w));});
test('stale plan source/runtime and selector invalidation refuse',async()=>{const x=setup(),w=await x.observe(),c=x.compose(w);for(const b of [{...c.bindings,runtimeIdentity:'f'.repeat(64)},{...c.bindings,sourceIdentity:{...x.identity,ino:'999'}}])assert.throws(()=>x.selector.requireCurrent(c.token,b));x.selector.invalidate('closed');assert.throws(()=>x.selector.requireCurrent(c.token,c.bindings));});

test('async refresh retains current witness and revokes on real identity change',async()=>{const x=setup(),w=await x.observe(),c=x.compose(w);await x.witness.refreshNativeOrientation(w,c.bindings);x.identity.ino='100';await assert.rejects(x.witness.refreshNativeOrientation(w,c.bindings));assert.throws(()=>x.selector.requireCurrent(c.token,c.bindings));});
test('abort refresh revokes even matching old bindings',async()=>{const x=setup(),w=await x.observe(),c=x.compose(w),a=new AbortController();a.abort();await assert.rejects(x.witness.refreshNativeOrientation(w,c.bindings,{signal:a.signal}));assert.throws(()=>x.selector.requireCurrent(c.token,c.bindings));});
