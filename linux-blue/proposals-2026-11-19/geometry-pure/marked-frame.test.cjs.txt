'use strict';
const test=require('node:test'),assert=require('node:assert/strict');const m=require('./marked-frame.cjs');
function transform(b,fn){const out=Buffer.from(b),w=720,h=1080;for(let y=0;y<h;y++)for(let x=0;x<w;x++){const [sx,sy]=fn(x,y);out[y*w+x]=sx>=0&&sx<w&&sy>=0&&sy<h?b[Math.floor(sy)*w+Math.floor(sx)]:96;}return out;}
test('ideal deterministic endpoints, all three selected positions',()=>{for(const i of [19,50,81]){assert.deepEqual(m.frame(i),m.frame(i));m.compare(m.frame(i),m.frame(i,true));}});
test('bounded deterministic interior perturbations',()=>{const a=m.frame(19),b=m.frame(19,true);for(let i=0;i<720*1080;i++)b[i]+=(i%3)-1;m.compare(a,b);});
for(const [name,fn] of [['horizontal mirror',(x,y)=>[719-x,y]],['vertical mirror',(x,y)=>[x,1079-y]],['16 pixel crop and rescale',(x,y)=>[16+x*688/720,16+y*1048/1080]],['four pixel translation',(x,y)=>[x-4,y]],['nonuniform stretch',(x,y)=>[360+(x-360)/1.08,y]]])test(name+' refused',()=>assert.throws(()=>m.analyze(transform(m.frame(19,true),fn),true)));
test('missing and swapped corners refused',()=>{const b=m.frame(19,true);for(let y=24;y<96;y++)b.fill(96,y*720+24,y*720+96);assert.throws(()=>m.analyze(b,true));const c=m.frame(19,true);for(let y=24;y<96;y++){c.fill(80,y*720+24,y*720+96);c.fill(32,y*720+624,y*720+696);}assert.throws(()=>m.analyze(c,true));});
test('single edge and square displacement refused',()=>{for(const r of m.rectangles(true).slice(4)){const b=m.frame(19,true);for(let y=r.y;y<r.y+r.h;y++)b.fill(96,y*720+r.x,y*720+r.x+r.w);assert.throws(()=>m.analyze(b,true),r.name);}});
test('exact length and frame index bounds',()=>{assert.throws(()=>m.frame(-1));assert.throws(()=>m.frame(240));assert.throws(()=>m.analyze(Buffer.alloc(1)));});
test('capture exact three, rejects overflow, truncation and reuse',()=>{for(const out of [false,true]){const c=m.captureThree(out);for(const i of [19,50,81])c.push(m.frame(i,out));assert.equal(c.finish().length,3);assert.throws(()=>c.finish());const d=m.captureThree(out);assert.throws(()=>d.finish());const e=m.captureThree(out);assert.throws(()=>e.push(Buffer.alloc(16*1024*1024)));}});
test('producer awaits each sink, abort and failure stop further allocation',async()=>{let active=0,max=0,count=0;await m.streamFrames(async()=>{active++;max=Math.max(max,active);await Promise.resolve();active--;count++;},{count:3});assert.equal(count,3);assert.equal(max,1);const ac=new AbortController();count=0;await assert.rejects(m.streamFrames(async()=>{count++;ac.abort();},{count:3,signal:ac.signal}));assert.equal(count,1);count=0;await assert.rejects(m.streamFrames(async()=>{count++;throw Error('sink');},{count:3}));assert.equal(count,1);});
