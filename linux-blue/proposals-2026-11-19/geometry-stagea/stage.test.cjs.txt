'use strict';
const test=require('node:test'),assert=require('node:assert/strict'),fs=require('node:fs');
const driver=require('./driver.cjs'),g=require('./geometry-controls.cjs'),checks=require('./frozen-checks.cjs');
function plan(){const filters=['scale=w=720:h=1080:flags=lanczos+accurate_rnd:in_color_matrix=bt709:out_color_matrix=bt709:in_range=tv:out_range=tv','setsar=sar=1/1:max=2147483647','sidedata=mode=delete:type=DISPLAYMATRIX','trim=start_pts=9416:end_pts=41795','setpts=PTS-9416'];return {expectedOutput:{geometry:{output:{width:720,height:1080}},filters:{video:filters}},args:['-filter:v:0',filters.join(',')]};}
test('unbound entry refuses before runtime/source/owner access',async()=>{const saved=process.execArgv.splice(0);try{await assert.rejects(driver.runOnce(),/UNBOUND/);}finally{process.execArgv.push(...saved);}assert.equal(JSON.parse(fs.readFileSync(require.resolve('./BINDING.json'))).sourceRoot,null);});
test('single portrait exact16 ledger, no spare capacity or hidden generation',()=>{assert.equal(driver.SPECS.length,1);assert.deepEqual(driver.SPECS[0],g.SPEC);assert.equal(Object.values(driver.LEDGER).reduce((a,b)=>a+b),16);assert.equal(driver.POLICY.maxChildren,16);assert.equal(driver.LEDGER.collect,6);assert.equal(driver.LEDGER.inspect,2);assert(driver.POLICY.totalMs<600000);});
test('canonical filter projection removes only geometry, never guesses timing',()=>{const p=plan(),c=g.controls(p);assert.deepEqual(c.sourceIndices,[19,50,81]);assert.equal(c.frames,63);assert.deepEqual(c.referenceFilters,p.expectedOutput.filters.video.slice(3));const args=g.imageArgs('/source',0,c.referenceFilters);assert(args[args.indexOf('-vf')+1].startsWith(c.referenceFilters.join(',')));assert(args.includes('passthrough'));});
test('wrong dimensions, hidden transform and mismatched argv refuse',()=>{for(const mutate of [p=>p.expectedOutput.geometry.output.width=1080,p=>p.expectedOutput.filters.video.push('hflip'),p=>p.args[1]='trim=start=0']){const p=plan();mutate(p);assert.throws(()=>g.controls(p));}});
test('frozen source generator uses rawstdin and exact distinguishable chirps',()=>{const a=g.generation('/owned/portrait/source.mp4');assert.equal(a[a.indexOf('-video_size')+1],'960x1440');assert.equal(a[a.indexOf('-i')+1],'pipe:0');assert(a.some(x=>x.includes('523*t+5.5*t*t')&&x.includes('742*t+6.5*t*t')));assert.equal(a.at(-1),'/owned/portrait/source.mp4');});
test('temporal prospective contract remains originalAAC rules without measured labels',()=>{const r=checks.frozenQualifications('prospective','a'.repeat(64),{streams:[{codec_type:'video',time_base:'1/15360'},{codec_type:'audio',time_base:'1/48000'}]});assert.equal(r.timingQualification.aac.encoderDelaySamples,1024);assert.equal(r.timingQualification.aac.discardPaddingSamples,0);assert.equal(r.qualification.tolerances.sampleCount,1023);assert(r.frozenBeforeEncode);});
test('monotonic expiry cannot pass late final check',()=>{let t=0n;const check=driver.deadline(1,()=>t);check();t=1000000n;assert.throws(check,/deadline expired/);});
test('composition retains genuine branded collection and unpublished stage, no old runtime source manifest',()=>{const s=fs.readFileSync(require.resolve('./driver.cjs'),'utf8');assert(s.includes('familySelection:qualified'));assert(s.includes('sourcePolicy:qualified.sourcePolicy'));assert(s.includes('selector.verifyCollectedSource'));assert(s.includes('openCandidateCalibrationRuntime(manifest,{execute})'));assert(s.includes('publicationAuthorized:false,catalogActivated:false'));assert(!s.includes('852afc3c83c4475a393696e5b206a22e0f17231e'));assert(s.includes('await finalGuard(totalCurrent,()=>owner.check());return report;'));});
