import os, sys, json, time, stat, hashlib, subprocess, signal
from pathlib import Path
ROOT=Path(__file__).resolve().parent
WORK=ROOT.parent
PACKAGE=WORK/'UNBOUND_NATIVE083_PACKAGE'
AUDIT=WORK/'UNBOUND_NATIVE083_FILE_AUDIT.json'
AUDIT_SHA='UNBOUND'
PACKAGE_COUNT=None
MODES={'electron/ffmpeg-encoder-linux-candidate','electron/chrome-sandbox','electron/chrome_crashpad_handler','electron/resources/runtime/ffmpeg','electron/resources/runtime/ffprobe'}
INPUTS={
 'owned-input-ordinary':('fixture.mp4',5578679,'bc1edc43c90b14690d1a2a89986d4141b2719058a04e5901e4d6b3fdcd536d97'),
 'owned-input-measured':('fixture.mov',2615213,'48ebd66bf2597ee1e5efca0b471a480d04e8cd83639259fda3539a99201552ae'),
 'owned-input-remux':('fixture.mp4',5578679,'bc1edc43c90b14690d1a2a89986d4141b2719058a04e5901e4d6b3fdcd536d97'),
}
OUTPUTS=['owned-output-ordinary','owned-output-measured','owned-output-remux']
SCOPE='native083-three-route-same-owner-publication'
FLOOR=27*1024**3
MAX_BYTES=2*1024**3
MAX_LOG=8*1024**2

def digest(path):
 h=hashlib.sha256()
 with path.open('rb')as f:
  for block in iter(lambda:f.read(1024**2),b''):h.update(block)
 return h.hexdigest()
def regular(path,mode=None):
 s=path.lstat()
 assert stat.S_ISREG(s.st_mode) and s.st_uid==os.getuid() and s.st_nlink==1 and path.resolve()==path
 if mode is not None:assert stat.S_IMODE(s.st_mode)==mode
 return s

def verify_package():
 assert PACKAGE_COUNT is not None and AUDIT_SHA!='UNBOUND','Package audit binding required'
 assert digest(AUDIT)==AUDIT_SHA
 a=json.loads(AUDIT.read_text());assert a['outputRoot']==str(PACKAGE) and len(a['identityRows'])==PACKAGE_COUNT
 expected={r['path'] for r in a['identityRows']}
 actual=set()
 for f in PACKAGE.rglob('*'):
  assert not f.is_symlink()
  if f.is_file():actual.add(str(f.relative_to(PACKAGE)))
 assert actual==expected
 for r in a['identityRows']:
  f=PACKAGE/r['path'];s=regular(f,0o755 if r['path']in MODES else 0o600)
  assert s.st_size==r['bytes'] and digest(f)==r['sha256']
 return True

def verify_inputs():
 for name,(filename,size,sha) in INPUTS.items():
  d=ROOT/name;assert d.resolve()==d and d.is_dir()
  assert {f.name for f in d.iterdir()}=={filename}
  f=d/filename;assert regular(f).st_size==size and digest(f)==sha
 return True

def limits():
 st=os.statvfs(ROOT);assert st.f_bavail*st.f_frsize>=FLOOR,'Free-space floor'
 total=0
 for d in [*INPUTS,*OUTPUTS,'owned-profile']:
  root=ROOT/d;assert root.resolve()==root and root.is_dir()
  for f in root.rglob('*'):
   # Chromium creates owned profile Singleton symlinks/socket endpoints. Never
   # traverse or open those; input/output links remain forbidden.
   s=f.lstat()
   if stat.S_ISLNK(s.st_mode):assert d=='owned-profile';continue
   if stat.S_ISREG(s.st_mode):assert s.st_uid==os.getuid();total+=s.st_size
   else:assert stat.S_ISDIR(s.st_mode) or (d=='owned-profile' and stat.S_ISSOCK(s.st_mode))
 assert total<=MAX_BYTES,'Observed owned-tree ceiling'
 for n in ['APP.stdout','APP.stderr']:
  f=ROOT/n
  if f.exists():assert regular(f).st_size<=MAX_LOG,'Observed log ceiling'
 return total

def group_rows(group):
 rows=[]
 for f in Path('/proc').iterdir():
  if not f.name.isdigit():continue
  try:
   if f.stat().st_uid!=os.getuid():continue
   v=(f/'stat').read_text();fields=v[v.rfind(')')+2:].split()
   if int(fields[2])==group and int(fields[3])==group:
    rows.append({'pid':int(f.name),'startTicks':int(fields[19]),'state':fields[0]})
  except (FileNotFoundError,ProcessLookupError):continue
 return rows

def leader_identity(pid):
 f=Path('/proc')/str(pid);assert f.stat().st_uid==os.getuid()
 text=(f/'stat').read_text();v=text[text.rfind(')')+2:].split()
 assert int(v[2])==pid and int(v[3])==pid,'Leader session/group changed'
 return (pid,int(v[19]))

def require_owner(identity):
 assert identity is not None and leader_identity(identity[0])==identity,'Leader identity unavailable/changed'

def exited_unreaped(pid):
 return os.waitid(os.P_PID,pid,os.WEXITED|os.WNOHANG|os.WNOWAIT) is not None

def loader_guard():
 assert not os.environ.get('NODE_OPTIONS') and not any(v for k,v in os.environ.items()if k.startswith('LD_'))
 for name in ['ELECTRON_RUN_AS_NODE','ELECTRON_FORCE_IS_PACKAGED','ELECTRON_DISABLE_SANDBOX']:
  assert name not in os.environ,'Electron execution override refused'

def write_once(name,value):
 with (ROOT/name).open('x')as f:json.dump(value,f,indent=2);f.write('\n')

def main():
 assert sys.argv[1:]==['--execute-reviewed-once']
 assert sys.platform=='linux' and os.uname().machine=='x86_64'
 assert os.environ.get('DISPLAY'),'Use the real desktop session, not a guessed display'
 loader_guard()
 gate=json.loads((ROOT/'LEAD_RELEASE.json').read_text())
 assert gate['scope']==SCOPE and gate['approved'] is True and gate['independentReviewApproved'] is True and gate['preservationVerified'] is True
 assert gate['supervisorSha256']==digest(Path(__file__).resolve()) and gate['fileAuditSha256']==AUDIT_SHA
 for key in ['reviewReceipt','preservationReceipt']:
  pin=gate[key];assert digest(Path(pin['path']))==pin['sha256']
 verify_package();verify_inputs();limits()
 for name in ['owned-profile',*OUTPUTS]:assert not any((ROOT/name).iterdir())
 write_once('ATTEMPT_STARTED.json',{'scope':SCOPE,'atUtc':time.time(),'package':str(PACKAGE)})
 result={'scope':SCOPE,'status':'starting','sampledOnly':True,'terminationRequested':False,'errors':[]};proc=None;owner=None;observed={};started=time.monotonic()
 try:
  with (ROOT/'APP.stdout').open('xb')as out,(ROOT/'APP.stderr').open('xb')as err:
   proc=subprocess.Popen([str(PACKAGE/'electron/ffmpeg-encoder-linux-candidate'),'--linux-unified-candidate','--user-data-dir='+str(ROOT/'owned-profile')],cwd=str(PACKAGE/'electron'),stdout=out,stderr=err,start_new_session=True)
   result['leaderPid']=proc.pid;owner=leader_identity(proc.pid)
   result['leaderIdentity']=owner
   while not exited_unreaped(proc.pid):
    require_owner(owner)
    for r in group_rows(proc.pid):observed[(r['pid'],r['startTicks'])]=r
    limits();assert time.monotonic()-started<600,'Native lifetime exceeded 10 minutes'
    time.sleep(.5)
   require_owner(owner)
   result['leaderExitObservedUnreaped']=True
 except BaseException as e:result['errors'].append({'phase':'execution','type':type(e).__name__,'message':str(e)[:1000]})
 finally:
  if proc is not None:
   try:
    require_owner(owner)
    live=[r for r in group_rows(proc.pid)if r['state']!='Z']
    if live:
     result['terminationRequested']=True
     for sig,seconds in [(signal.SIGTERM,5),(signal.SIGKILL,3)]:
      require_owner(owner)
      if not [r for r in group_rows(proc.pid)if r['state']!='Z']:break
      os.killpg(proc.pid,sig)
      until=time.monotonic()+seconds
      while time.monotonic()<until and [r for r in group_rows(proc.pid)if r['state']!='Z']:time.sleep(.1)
    require_owner(owner)
    result['remainingGroupRows']=group_rows(proc.pid)
    result['closureObserved']=not any(r['state']!='Z'for r in result['remainingGroupRows'])
    if result['closureObserved']:
     proc.wait(timeout=1) # Reap only after identity-bound group closure.
     result['exitCode']=proc.returncode
     if proc.returncode!=0:result['errors'].append({'phase':'exit','message':'Native app exited nonzero'})
   except BaseException as e:
    result['closureObserved']=False
    result['errors'].append({'phase':'closure-unverifiable','type':type(e).__name__,'message':str(e)[:1000]})
  for phase,fn in [('package',verify_package),('inputs',verify_inputs),('storage',limits)]:
   try:result[phase+'Verified']=fn()
   except BaseException as e:result['errors'].append({'phase':phase,'type':type(e).__name__,'message':str(e)[:1000]})
  result['sampledIdentities']=list(observed.values());result['elapsedSeconds']=time.monotonic()-started
  result['status']='closed-awaiting-evidence-audit'if not result['errors'] and result.get('closureObserved') and not result['terminationRequested'] else 'failed-or-terminated-retained'
  write_once('CLOSURE.json',result)
 if result['status']!='closed-awaiting-evidence-audit':raise SystemExit(1)
if __name__=='__main__':main()
