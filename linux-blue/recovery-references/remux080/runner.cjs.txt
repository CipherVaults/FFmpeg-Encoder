async function pinned(p,pin){const s=await fs.lstat(p);assert(s.isFile()&&!s.isSymbolicLink()&&s.nlink===1&&s.size===pin.bytes&&s.size<=16*1024*1024);assert.equal(await fs.realpath(p),p);const b=await fs.readFile(p);assert.equal(b.length,pin.bytes);assert.equal(crypto.createHash('sha256').update(b).digest('hex'),pin.sha256);return b;}
async function write(name,value){await fs.writeFile(path.join(ROOT,name),JSON.stringify(value,null,2)+'\n',{flag:'wx',mode:0o600});}
function cancellationConclusion(item,requested){
 if(item.status==='cancelled'){
  assert(requested,'Unrequested cancellation is not this test');
  assert.notEqual(item.outputDisposition?.published,true);assert(!item.outputAccess);
  return 'cancelled';
 }
 assert.equal(item.status,'success','Cancellation attempt failed rather than completing too quickly');
 assert.equal(item.outputDisposition?.published,true,'Successful race must retain published output');
 return 'inconclusive-too-quick';
}
